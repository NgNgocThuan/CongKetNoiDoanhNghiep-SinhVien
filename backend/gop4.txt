from app.core.enums import ApplicationStatus
from app.core.exceptions import BadRequestError, NotFoundError, ConflictError

from app.services.notification_service import create_notification

# Quy tắc chuyển trạng thái hợp lệ
_VALID_TRANSITIONS = {
    ApplicationStatus.APPLIED: {ApplicationStatus.SCREENING, ApplicationStatus.REJECTED, ApplicationStatus.WITHDRAWN},
    ApplicationStatus.SCREENING: {ApplicationStatus.SHORTLISTED, ApplicationStatus.REJECTED, ApplicationStatus.WITHDRAWN},
    ApplicationStatus.SHORTLISTED: {ApplicationStatus.INTERVIEW, ApplicationStatus.REJECTED, ApplicationStatus.WITHDRAWN},
    ApplicationStatus.INTERVIEW: {ApplicationStatus.OFFER, ApplicationStatus.REJECTED, ApplicationStatus.WITHDRAWN},
    ApplicationStatus.OFFER: {ApplicationStatus.HIRED, ApplicationStatus.REJECTED, ApplicationStatus.WITHDRAWN},
}


def _get_owned_application(db: Session, company: Company, application_id: int) -> Application:
    stmt = (
        select(Application)
        .join(Application.job)
        .where(Application.id == application_id, Job.company_id == company.id)
        .options(
            selectinload(Application.job),
            selectinload(Application.student).selectinload(Student.user),
            selectinload(Application.cv),
            selectinload(Application.status_logs),
        )
    )
    app = db.scalars(stmt).first()
    if not app:
        raise NotFoundError("Không tìm thấy đơn ứng tuyển thuộc tin của doanh nghiệp")
    return app


def list_applications(
    db: Session,
    company: Company,
    job_id: int,
    params: PageParams,
    status: Optional[ApplicationStatus] = None,
    major_id: Optional[int] = None,
    min_gpa: Optional[float] = None,
    skill_id: Optional[int] = None,
    min_experience_months: Optional[int] = None,
) -> dict:
    """Danh sách đơn theo job_id; lọc ngành, GPA, kỹ năng, kinh nghiệm; hỗ trợ Kanban."""
    # Kiểm tra job có thuộc công ty không
    job = db.get(Job, job_id)
    if not job or job.company_id != company.id:
        raise NotFoundError("Không tìm thấy tin tuyển dụng")

    stmt = (
        select(Application)
        .join(Application.student)
        .where(Application.job_id == job_id)
        .options(
            selectinload(Application.student).selectinload(Student.user),
            selectinload(Application.student).selectinload(Student.major),
            selectinload(Application.cv),
        )
    )

    if status is not None:
        stmt = stmt.where(Application.current_status == status)
    if major_id is not None:
        stmt = stmt.where(Student.major_id == major_id)
    if min_gpa is not None:
        stmt = stmt.where(Student.gpa >= min_gpa)
    if min_experience_months is not None:
        stmt = stmt.where(Student.experience_months >= min_experience_months)
    if skill_id is not None:
        from app.models import StudentSkill
        stmt = stmt.where(Application.student_id.in_(select(StudentSkill.student_id).where(StudentSkill.skill_id == skill_id)))

    stmt = stmt.order_by(Application.created_at.desc())
    return paginate(db, stmt, params)


def get_application(db: Session, company: Company, application_id: int) -> Application:
    """Hồ sơ sinh viên + CV snapshot chỉ của đơn thuộc tin công ty mình."""
    return _get_owned_application(db, company, application_id)


def log_status_change(
    db: Session,
    background_tasks: BackgroundTasks,
    company: Company,
    application_id: int,
    new_status: ApplicationStatus,
    note: Optional[str] = None,
) -> Application:
    """Đổi trạng thái đơn (Kanban), kiểm tra hợp lệ, ghi log, gửi thông báo + email."""
    app = _get_owned_application(db, company, application_id)
    old_status = app.current_status

    if old_status in (ApplicationStatus.HIRED, ApplicationStatus.REJECTED, ApplicationStatus.WITHDRAWN):
        raise ConflictError("Đơn đã ở trạng thái kết thúc, không thể thay đổi", code="APPLICATION_CLOSED")

    if new_status != old_status:
        allowed = _VALID_TRANSITIONS.get(old_status, set())
        # REJECTED có thể chuyển từ bất kỳ bước nào chưa kết thúc
        if new_status != ApplicationStatus.REJECTED and new_status not in allowed:
            raise BadRequestError(f"Không thể chuyển trạng thái từ {old_status.value} sang {new_status.value}")

    app.current_status = new_status

    # Ghi log trạng thái
    status_log = ApplicationStatusLog(
        application_id=app.id,
        from_status=old_status,
        to_status=new_status,
        note=note,
    )
    db.add(status_log)
    db.commit()
    db.refresh(app)

    # Gửi thông báo và email cho sinh viên
    student_user = app.student.user
    if student_user:
        title = "Cập nhật trạng thái đơn ứng tuyển"
        content = f"Đơn ứng tuyển vào vị trí '{app.job.title}' đã chuyển sang trạng thái: {new_status.value}."
        if note:
            content += f" Ghi chú: {note}"

        create_notification(
            db,
            user_id=student_user.id,
            type="APPLICATION_STATUS",
            title=title,
            content=content,
            ref_type="APPLICATION",
            ref_id=app.id,
        )

        email_body = f"<p>Xin chào {app.student.full_name},</p><p>{content}</p>"
        queue_email(
            db,
            background_tasks,
            to_email=student_user.email,
            subject="Cập nhật trạng thái đơn ứng tuyển",
            body_html=email_body,
            type="STATUS_CHANGED",
            user_id=student_user.id,
            ref_type="APPLICATION",
            ref_id=app.id,
        )

    return app


def get_history(db: Session, company: Company, application_id: int) -> list[ApplicationStatusLog]:
    """Lịch sử xử lý đơn ứng tuyển."""
    app = _get_owned_application(db, company, application_id)
    return app.status_logs

"""Nghiệp vụ: Lời mời ứng tuyển. Người phụ trách: Khôi."""
from datetime import datetime, timedelta
from typing import Optional
from fastapi import BackgroundTasks
from sqlalchemy import select, and_, update
from sqlalchemy.orm import Session

from app.core.enums import InvitationStatus
from app.core.exceptions import BadRequestError, NotFoundError, ConflictError
from app.models import Invitation, Job, Student, Company
from app.services.email_service import queue_email
from app.services.notification_service import create_notification
from app.schemas.common import PageParams
from app.utils.pagination import paginate


def send_invitation(
    db: Session,
    background_tasks: BackgroundTasks,
    company: Company,
    job_id: int,
    student_id: int,
    message: Optional[str] = None,
) -> Invitation:
    """Doanh nghiệp mời sinh viên (chỉ mời sinh viên is_open_to_work = True; duy nhất theo tin và sinh viên)."""
    job = db.get(Job, job_id)
    if not job or job.company_id != company.id:
        raise NotFoundError("Không tìm thấy tin tuyển dụng")

    student = db.get(Student, student_id)
    if not student:
        raise NotFoundError("Không tìm thấy hồ sơ sinh viên")

    if not student.is_open_to_work:
        raise BadRequestError("Sinh viên hiện không bật trạng thái tìm việc (is_open_to_work)")

    existing = db.scalar(select(Invitation).where(Invitation.job_id == job_id, Invitation.student_id == student_id))
    if existing:
        raise ConflictError("Đã gửi lời mời cho sinh viên này với tin tuyển dụng này rồi", code="DUPLICATE_INVITATION")

    invitation = Invitation(
        job_id=job_id,
        student_id=student_id,
        company_id=company.id,
        message=message,
        status=InvitationStatus.PENDING,
    )
    db.add(invitation)
    db.commit()
    db.refresh(invitation)

    student_user = student.user
    if student_user:
        title = f"Lời mời ứng tuyển từ {company.name}"
        content = f"Công ty {company.name} mời bạn ứng tuyển vị trí: {job.title}."
        if message:
            content += f" Lời nhắn: {message}"

        create_notification(
            db,
            user_id=student_user.id,
            type="INVITATION",
            title=title,
            content=content,
            ref_type="INVITATION",
            ref_id=invitation.id,
        )

        email_body = f"<p>Xin chào {student.full_name},</p><p>{content}</p>"
        queue_email(
            db,
            background_tasks,
            to_email=student_user.email,
            subject="Lời mời ứng tuyển việc làm",
            body_html=email_body,
            type="INVITATION",
            user_id=student_user.id,
            ref_type="INVITATION",
            ref_id=invitation.id,
        )

    return invitation


def list_for_company(db: Session, company: Company, params: PageParams) -> dict:
    stmt = select(Invitation).where(Invitation.company_id == company.id).order_by(Invitation.created_at.desc())
    return paginate(db, stmt, params)


def list_for_student(db: Session, student: Student, params: PageParams) -> dict:
    stmt = select(Invitation).where(Invitation.student_id == student.id).order_by(Invitation.created_at.desc())
    return paginate(db, stmt, params)


def decline(db: Session, student: Student, invitation_id: int) -> Invitation:
    invitation = db.get(Invitation, invitation_id)
    if not invitation or invitation.student_id != student.id:
        raise NotFoundError("Không tìm thấy lời mời ứng tuyển")

    if invitation.status != InvitationStatus.PENDING:
        raise ConflictError("Lời mời đã được xử lý trước đó", code="INVITATION_PROCESSED")

    invitation.status = InvitationStatus.DECLINED
    db.commit()
    db.refresh(invitation)
    return invitation


def expire_old_invitations(db: Session) -> int:
    """Hết hạn các lời mời cũ quá hạn (nếu cần chạy định kỳ)."""
    threshold = datetime.utcnow() - timedelta(days=30)
    result = db.execute(
        update(Invitation)
        .where(Invitation.status == InvitationStatus.PENDING, Invitation.created_at < threshold)
        .values(status=InvitationStatus.EXPIRED)
    )
    db.commit()
    return result.rowcount or 0

from pydantic import BaseModel, ConfigDict, Field
